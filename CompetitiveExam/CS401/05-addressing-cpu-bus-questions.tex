\noindent Each question carries 1--2 marks. MCQ = single correct option; NAT = Numerical Answer Type.

\begin{enumerate}
\item[\textbf{Q1}] \textit{[GATE CS 1996 Q1.16, verified --- GATEOverflow V3 p.29]} \textbf{[MCQ, 1 mark]} Relative mode of addressing is most relevant to writing:
\begin{enumerate}
  \item[(A)] Co-routines
  \item[(B)] Position-independent code
  \item[(C)] Shareable code
  \item[(D)] Interrupt handlers
\end{enumerate}

\item[\textbf{Q2}] \textit{[GATE CS 1998 Q1.19, verified --- GATEOverflow V3 p.29]} \textbf{[MCQ, 1 mark]} Which of the following addressing modes permits relocation without any change whatsoever in the code?
\begin{enumerate}
  \item[(A)] Indirect addressing
  \item[(B)] Indexed addressing
  \item[(C)] Base register addressing
  \item[(D)] PC-relative addressing
\end{enumerate}

\item[\textbf{Q3}] \textit{[GATE CS 1999 Q2.23, verified --- GATEOverflow V3 p.29]} \textbf{[MCQ, 2 marks]} A certain processor supports only the immediate and the direct addressing modes. Which of the following programming-language features cannot be implemented on this processor?
\begin{enumerate}
  \item[(A)] Pointers
  \item[(B)] Arrays
  \item[(C)] Records
  \item[(D)] Recursive procedures with local variables
\end{enumerate}

\item[\textbf{Q4}] \textit{[GATE CS 2000 Q1.10, verified --- GATEOverflow V3 p.29]} \textbf{[MCQ, 1 mark]} The most appropriate matching for the following pairs is:
\begin{center}
\small
X: Indirect addressing \quad 1: Loops \\
Y: Immediate addressing \quad 2: Pointers \\
Z: Autodecrement addressing \quad 3: Constants
\end{center}
\begin{enumerate}
  \item[(A)] X--3, Y--2, Z--1
  \item[(B)] X--1, Y--3, Z--2
  \item[(C)] X--2, Y--3, Z--1
  \item[(D)] X--3, Y--1, Z--2
\end{enumerate}

\item[\textbf{Q5}] \textit{[GATE CS 2002 Q1.24, verified --- GATEOverflow V3 p.30]} \textbf{[MCQ, 1 mark]} In the absolute addressing mode:
\begin{enumerate}
  \item[(A)] the operand is inside the instruction
  \item[(B)] the address of the operand is inside the instruction
  \item[(C)] the register containing the address of the operand is specified inside the instruction
  \item[(D)] the location of the operand is implicit
\end{enumerate}

\item[\textbf{Q6}] \textit{[GATE CS 2004 Q20, verified --- GATEOverflow V3 p.30]} \textbf{[MCQ, 1 mark]} Which of the following addressing modes are suitable for program relocation at run time?
\begin{enumerate}
  \item[(A)] Absolute addressing only
  \item[(B)] Based and relative addressing
  \item[(C)] Indexed addressing only
  \item[(D)] Indirect addressing only
\end{enumerate}

\item[\textbf{Q7}] \textit{[Original, GATE-pattern]} \textbf{[MCQ, 2 marks]} In a single-bus CPU, the instruction $R1 \gets R2 + R3$ is implemented over three control steps using two hidden registers. The reason the same transfer takes only \emph{one} step on a three-bus organization is that:
\begin{enumerate}
  \item[(A)] the ALU is faster on a three-bus design
  \item[(B)] three buses can carry both operands and the result in the same cycle without a shared-path bottleneck
  \item[(C)] the register file has more registers on a three-bus design
  \item[(D)] control logic is simpler on a three-bus design
\end{enumerate}

\end{enumerate}
